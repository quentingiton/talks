	\begin{frame}[t]{What is the score function?}
		\begin{itemize}[<+->]
			\item\textbf{Forward process:} Ornstein--Uhlenbeck (OU) process:
			\begin{equation*}
				\d X_t = - X_t\d t+\sqrt{2}\d\omega_t,\quad  X_0 \sim \mu,\quad t\in[0,T].
			\end{equation*}
			Let $\rho_t :=\mathrm{Law}( X_t)$. Then $\rho_t$ solves the Fokker--Planck equation
			\begin{equation}\label{eq:FPE}
				\partial_t\rho_t(x) = \Delta\rho_t(x) + \mathrm{div}(\rho_t(x)x).\tag*{\rust{($\star$)}}
			\end{equation}
			whose stationary solution is $\gamma\propto e^{-\|\cdot\|^2/2}$.
			
			\item \textbf{Macroscopic POV:} Equation \ref{eq:FPE} is the $2$-Wasserstein Gradient Flow of the relative entropy $\mathcal{H}(\rho\vert\gamma):=\int \frac{\|x\|^2}{2}\rho+\int\rho\ln{\rho}$. It means \ref{eq:FPE} can be written as
			\begin{equation*}
				\partial_t\rho_t(x) = -\nabla\cdot\left( \rho_t(x)v_t(x) \right),
			\end{equation*}
			where $v_t(x) = -x-\nabla\ln{\rho_t(x)}$.
		\end{itemize}
	\end{frame}
	
	
	
	\begin{frame}[t]{Using OT to learn the score 1/2}
		\begin{itemize}[<+->]
			\item Given a time-step $\tau>0$, such a gradient flow can be discretised in time using the \textbf{JKO scheme} (or Implicit Euler scheme, or Proximal step)
			\begin{equation*}
				\left\{ 
					\begin{array}{l}
						\rho_0 = \mu \\
						\displaystyle \rho_{n+1}^\tau := \argmin{} \mathcal{H}(\rho\vert\gamma)+\frac{1}{2\tau}W_2(\rho_n^\tau,\rho)^2
					\end{array}
				 \right.
			\end{equation*} 
			
			\item \textbf{At step $n$:} We show that
			\begin{equation*}
				\purp{\rho_{n+1}^\tau}\propto\exp{-(\varphi^\star)^{c/2\tau}}\d\purp{\gamma}
			\end{equation*}
			where $\varphi^\star$ is a Kantorovich potential satisfying
			\begin{equation*}
				\varphi^\star\in\argmin{\varphi} \tilde J(\varphi) := \ln{\int e^{-\varphi^{c/2\tau}}\d\purp{\gamma}} - \int\varphi\d\rust{\rho_n^\tau}.
			\end{equation*}
		\end{itemize}
	\end{frame}

	\begin{frame}[t]{General contributions}
		\begin{myprop*}{\itshape (G.)}

	\begin{frame}[t]{Stochastic gradient descent 1/3}
		\begin{itemize}[<+->]
			\item $\displaystyle J(\varphi) = \mathbb{E}_{\purp{y}\sim\purp{\gamma}}\left[ \exp{-\varphi^{c/2\tau}(\purp{y})}-\int\varphi(\rust{x})\rust{\rho_n^\tau(}\d\rust{x)} \right]$
			\vspace{\baselineskip}
			\item Choice of parametrisation: semi-discrete optimal transport using
			\begin{equation*}
				\rust{\hat\rho_n^\tau} = \frac{1}{N}\sum_{i=1}^N \delta_{\rust{\hat x_i^{(n)}}} \leadsto 
				\left\{ 
					\begin{array}{l}
						\varphi(x;\theta) = \theta^\top\varphi(x) \\
						\varphi(x) = (\varphi_1(x),\dots, \varphi_N(x)) \\
						\varphi_i(x) = \mathds{1}_{\{x = \hat x_i^{(n)}\}}
					\end{array}\right.
			\end{equation*}
			\vspace{\baselineskip}
			\item $\displaystyle G(\theta) := J(\varphi(\cdot;\theta)) = \mathbb{E}_{\purp{y}\sim\purp{\gamma}}\left[ \exp{-\min_{1\leq i\leq N} \left\{ \frac{c(\rust{\hat x_i^{(n)}},\purp{y})}{2\tau}-\theta_i \right\}}-\frac{1}{N}\sum_{i=1}^N\theta_i \right]$
		\end{itemize}
	\end{frame}
	
	
	
	\begin{frame}[t]{Stochastic gradient descent 2/3}

		\begin{equation*}
			\left\{
			\begin{array}{l}
				\text{pick } \purp{y}\sim \purp{\gamma} \\
				\text{let } i :=\mathrm{arg}\,\mathrm{min}_j\  c_\tau(\rust{\hat x_j^{(n)}},\purp{y})-\theta_j \\
				\text{define } e_i \text{ as the $i$-th canonical basis vector of $\mathbb{R}^N$} \\
				\theta^{(k+1)} := \mathrm{proj}_{B(0,R_\mathrm{proj})}\left( \theta^{(k)} -  \alpha_k \smash{\underbrace{e^{\theta_i^{(k)}-c_\tau(\rust{\hat x_j^{(n)}},\purp{y})}}_{\text{A priori \emph{unbounded}!}}}e_i+ \frac{\alpha_k}{N}\mathds{1} \right)

			\right.
		\end{equation*}

	\begin{frame}[t]{Stochastic gradient descent 3/3}
		\begin{mythm*}{\itshape (G.)}
			\begin{itemize}
				\item $\alpha_k = \frac{C}{\sqrt{k}},\quad \hat\theta_K := \frac{1}{K}\sum_{k=0}^K \theta^{(k)}$

		\end{mythm*}

		\end{itemize}

	\begin{frame}[t]{More on the barycentric map}
		\begin{itemize}[<+->]
