			\item $p$-Wasserstein distance:
			\begin{equation*}
				W_p(\rust{\mu},\purp{\nu})^p = \inf_{\pi\in\Pi(\rust{\mu},\purp{\nu})} \int d(\rust{x},\purp{y})^p \gamma(\mathrm{d}\rust{x},\mathrm{d}\purp{y})
			\end{equation*}
			\item Kantorovich duality
			\begin{equation*}
				\inf_{\pi\in\Pi(\rust{\mu},\purp{\nu})} \int c(\rust{x},\purp{y}) \gamma(\mathrm{d}\rust{x},\mathrm{d}\purp{y}) = \sup_{\phi} \int \phi(\rust{x})\rust{\mu(}\mathrm{d}\rust{x)}+\int\phi^c(\purp{y})\purp{\nu(}\mathrm{d}\purp{y)},
			\end{equation*}
			where $\phi^c(\purp{y}) := \inf_{\rust{x}} c(\rust{x},\purp{y})-\phi(\rust{x})$.
		\end{itemize}

			\item Kantorovich duality
			\begin{equation*}
				\inf_{T\::\: T_\#\textcolor{brandAccent}{\mu}=\textcolor{brandPurple}{\nu}} \int_{\mathbb{R}^d} c(\textcolor{brandAccent}{x},T(\textcolor{brandAccent}{x}))\textcolor{brandAccent}{\mu(}\mathrm{d}\textcolor{brandAccent}{x)} = \sup_{\phi} \int \phi(\rust{x})\rust{\mu(}\mathrm{d}\rust{x)}+\int\phi^c(\purp{y})\purp{\nu(}\mathrm{d}\purp{y)}
			\end{equation*}
			\item If an optimal transport map $T^\star$ exists, there also exists an optimal Kantorovich potential $\phi^\star$ and
			\begin{equation*}
				\purp{y}=T^\star(\rust{x}) = \nabla\left( \frac{\|\cdot\|^2}{2}-\phi^\star \right)(\rust{x})=\rust{x}-\nabla\phi^\star(\rust{x}),
			\end{equation*}
			hence $-\nabla\phi^\star(\rust{x})=T^\star(\rust{x})-\rust{x}=\purp{y}-\rust{x}$ is the \textbf{displacement} of the particle $\rust{x}$.
		\end{itemize}

	\begin{frame}{What is the score function?}
		\begin{itemize}[<+->]
			\item\textbf{Forward process:} Ornstein--Uhlenbeck (OU) process:
			\begin{equation*}
				\d \vec{X}_t = -\vec X_t\d t+\sqrt{2}\d\omega_t,\quad \vec X_0 \sim \mu,\quad t\in[0,T].
			\end{equation*}
			Let $\vec\rho_t :=\mathrm{Law}(\vec X_t)$ and $\gamma\propto e^{-\|\cdot\|^2/2}$. Then $\vec\rho_t$ solves the Fokker--Planck equation
			\begin{equation*}
				\partial_t\vec\rho_t(x) = \Delta\vec\rho_t(x) + \mathrm{div}(\vec\rho_t(x)x).
			\end{equation*}
			whose stationary solution is $\gamma$. The process $(\vec X_t)_t$ is a \textbf{diffusive} process.
			
			\item \textbf{Backward process:} $\cev{X}_s := \vec X_{T-s}$ for $s\in[0,T]$. If $(\vec X_t)_{t\in[0,T]}$ is an OU process, then its time reversal $(\cev X_s)_{s\in[0,T]}$ solves
			\begin{equation*}
				\d\cev X_s = \Big[\cev X_s+2\underbrace{\nabla\ln{\vec\rho_{T-s}(\cev X_s)}}_{\textbf{The score function}}\Big]\d s + \sqrt{2}\d\omega_s,\quad \cev X_0 \sim \vec\rho_T.
			\end{equation*}
		\end{itemize}
	\end{frame}
	
	
	
	\begin{frame}{Using OT to learn the score 1/2}
		\begin{itemize}[<+->]
			\item The Fokker--Planck equation is the $2$-Wasserstein Gradient Flow (WGF) of the relative entropy $\mathcal{H}(\rho\vert\gamma):=\int V\rho+\int\rho\ln{\rho}$. That is:
			\begin{equation*}
				\partial_t\rho_t(x) = \Delta\rho_t(x) + \mathrm{div}(\rho_t(x)x) = -\nabla\cdot\left( \rho_t(x)v_t(x) \right),
			\end{equation*}
			where $v_t(x) = -x-\nabla\ln{\rho_t(x)}$.
			
			\vspace{\baselineskip}
			
			\item Given a time-step $\tau>0$, such a gradient flow can be discretized in time using the \textbf{JKO scheme}
			\begin{equation*}
				\left\{ 
					\begin{array}{l}
						\rho_0 = \mu \\
						\rho_{n+1}^\tau := \argmin{} \mathcal{H}(\rho\vert\gamma)+\frac{1}{2\tau}W_2(\rho_n^\tau,\rho)^2
					\end{array}
				 \right.
			\end{equation*} 
		\end{itemize}
	\end{frame}
	
	
	
	\begin{frame}{Using OT to learn the score 2/2}
		\begin{itemize}[<+->]
			\item Solving
			\begin{equation*}
				\inf_\rho \mathcal{H}(\rho\vert\gamma)+\frac{1}{2\tau}W_2(\rho_n^\tau,\rho)^2
			\end{equation*}
			amounts to finding the optimal transport map between
			\begin{equation*}
				\rust{\rho_n^\tau}\qquad \text{and}\qquad \purp{\rho_{n+1}^\tau}\propto\exp{-\phi^{c/2\tau}}\d\gamma,
			\end{equation*}
			where $\phi$ solves
			\begin{equation*}
				\inf_\phi \int e^{-\phi^{c/2\tau}}\d\gamma - \int\phi\d\rust{\rho_n^\tau}.
			\end{equation*}
			
			\item The optimal potential $\phi^\star$ satisfies
			\begin{equation*}
				-\nabla\phi^\star(\rust{x}) = \nabla(\phi^\star)^{c/2\tau}(\purp{y}) = \purp{y}-\rust{x}.
			\end{equation*}
			
			\item The displacement of the particle $\rust{x}$ being $v_t(\rust{x})$, we obtain
			\begin{equation*}
				-\nabla\phi^\star(\rust{x}) = v_t(\rust{x}) = -x-\nabla\ln{\rho_{n+1}^\tau(\rust{x})} \iff \boxed{\nabla\ln{\rho_{n+1}^\tau(\rust{x})} = \rust{x}+\nabla\phi^\star(\rust{x})}.
			\end{equation*}
		\end{itemize}
	\end{frame}
