	\begin{frame}[t]{What is the score function?}
		\begin{itemize}[<+->]
			\item\textbf{Forward process:} Ornstein--Uhlenbeck (OU) process:
			\begin{equation*}
				\d \vec{X}_t = -\vec X_t\d t+\sqrt{2}\d\omega_t,\quad \vec X_0 \sim \mu,\quad t\in[0,T].
			\end{equation*}
			Let $\vec\rho_t :=\mathrm{Law}(\vec X_t)$ and $\gamma\propto e^{-\|\cdot\|^2/2}$. Then $\vec\rho_t$ solves the Fokker--Planck equation
			\begin{equation}\label{eq:FPE}
				\partial_t\vec\rho_t(x) = \Delta\vec\rho_t(x) + \mathrm{div}(\vec\rho_t(x)x).\tag*{\rust{($\star$)}}
			\end{equation}
			whose stationary solution is $\gamma$.
			
			\item \textbf{Macroscopic POV:} Equation \ref{eq:FPE} is the $2$-Wasserstein Gradient Flow of the relative entropy $\mathcal{H}(\rho\vert\gamma):=\int \frac{\|x\|^2}{2}\rho+\int\rho\ln{\rho}$. It means \ref{eq:FPE} can be written as
			\begin{equation*}
				\partial_t\rho_t(x) = -\nabla\cdot\left( \rho_t(x)v_t(x) \right),
			\end{equation*}
			where $v_t(x) = -x-\nabla\ln{\rho_t(x)}$.
		\end{itemize}
	\end{frame}
	
	
	
	\begin{frame}[t]{Using OT to learn the score 1/2}
		\begin{itemize}[<+->]
			\item Given a time-step $\tau>0$, such a gradient flow can be discretised in time using the \textbf{JKO scheme} (or Implicit Euler scheme, or Proximal step)
			\begin{equation*}
				\left\{ 
					\begin{array}{l}
						\rho_0 = \mu \\
						\displaystyle \rho_{n+1}^\tau := \argmin{} \mathcal{H}(\rho\vert\gamma)+\frac{1}{2\tau}W_2(\rho_n^\tau,\rho)^2
					\end{array}
				 \right.
			\end{equation*} 
			
			\item \textbf{At step $n$:} We show that
			\begin{equation*}
				\purp{\rho_{n+1}^\tau}\propto\exp{-\phi^{c/2\tau}}\d\purp{\gamma}
			\end{equation*}
			where $\phi$ is such that
			\begin{equation*}
				\inf_\phi \tilde J(\phi) := \ln{\int e^{-\phi^{c/2\tau}}\d\purp{\gamma}} - \int\phi\d\rust{\rho_n^\tau}.
			\end{equation*}
		\end{itemize}
	\end{frame}
	
	
	
	\begin{frame}[t]{Using OT to learn the score 2/2}
		Recall that:
		\begin{equation*}
			\purp{\rho_{n+1}^\tau}\propto\exp{-(\phi^\star)^{c/2\tau}}\d\purp{\gamma} \quad \text{where}\quad \phi^\star\in\argmin{\phi} \tilde J(\phi) := \ln{\int e^{-\phi^{c/2\tau}}\d\purp{\gamma}} - \int\phi\d\rust{\rho_n^\tau}
		\end{equation*}
		\begin{itemize}[<+->]
			\item The optimal potential $\phi^\star$ satisfies ($\alpha=\frac{1}{2\tau}$ here)
			\begin{equation*}
				\underset{\textbf{Displacement}}{\boxed{-\tau\nabla\phi^\star(\rust{x}) = \purp{y}-\rust{x}}}
			\end{equation*}
			
			\item The displacement of the particle $\rust{x}$ being $\tau v_t(\rust{x})$, we obtain
			\begin{equation*}
				-\tau\nabla\phi^\star(\rust{x}) = \tau v_t(\rust{x}) = -\tau (\rust{x}+\nabla\ln{\rho_{n+1}^\tau(\rust{x})}) \iff \boxed{\nabla\ln{\rho_{n+1}^\tau(\rust{x})} = \nabla\phi^\star(\rust{x})-\rust{x}}.
			\end{equation*}
		\end{itemize}
	\end{frame}

	\begin{frame}[t]{Stochastic gradient descent 1/2}
		\begin{itemize}[<+->]
			\item $\displaystyle J(\phi) = \mathbb{E}_{\purp{y}\sim\purp{\gamma}}\left[ \exp{-\phi^{c/2\tau}(\purp{y})}-\int\phi(\rust{x})\rust{\rho_n^\tau(}\d\rust{x)} \right]$
			\vspace{\baselineskip}
			\item Choice of parametrisation: semi-discrete optimal transport using
			\begin{equation*}
				\rust{\hat\rho_n^\tau} = \frac{1}{N}\sum_{i=1}^N \delta_{\rust{\hat x_i^{(n)}}} \leadsto 
				\left\{ 
					\begin{array}{l}
						\phi(x;\theta) = \theta^\top\Phi(x) \\
						\Phi(x) = (\Phi_1(x),\dots, \Phi_N(x)) \\
						\Phi_i(x) = \mathds{1}_{\{x = \hat x_i^{(n)}\}}
					\end{array}\right.
			\end{equation*}
			\vspace{\baselineskip}
			\item $\displaystyle G(\theta) = \mathbb{E}_{\purp{y}\sim\purp{\gamma}}\left[ \max_{1\leq i\leq N}\left\{\exp{\theta_i - \frac{c(\rust{\hat x_i^{(n)}},\purp{y})}{2\tau}}\right\}-\frac{1}{N}\sum_{i=1}^N\theta_i \right]$
		\end{itemize}
	\end{frame}
	
	
	
	\begin{frame}[t]{Stochastic gradient descent 2/2}
		Projected SGD algorithm $\big(c_\tau(\rust{x},\purp{y}):=\frac{c(\rust{x},\purp{y})}{2\tau}\big)$
		\begin{equation*}
			\left\{
			\begin{array}{l}
				\text{pick } \purp{y}\sim \purp{\gamma} \\
				\text{let } i :=\mathrm{arg}\,\mathrm{min}_j\  c_\tau(\rust{\hat x_j^{(n)}},\purp{y})-\theta_j \\
				\text{define } e_i \text{ as the $i$-th canonical basis vector of $\mathbb{R}^N$} \\
				\theta^{(k+1)} := \mathrm{proj}_{B(0,R_\mathrm{proj})}\left( \theta^{(k)} -  \alpha_ke^{\theta_i^{(k)}-c_\tau(\rust{\hat x_j^{(n)}},\purp{y})}e_i+ \frac{\alpha_k}{N}\mathds{1} \right)
			\end{array}
			\right.
		\end{equation*}
		
		\begin{mythm*}{\itshape (G.)}
			\begin{itemize}
				\item $\alpha_k = \frac{C}{\sqrt{k}},\quad \hat\theta_K := \frac{1}{K}\sum_{k=0}^K \theta^{(k)}$
			\end{itemize}
			Then, it holds that
			\begin{equation*}
				0\leq \mathbb E [G(\hat\theta_K)] - G(\theta^\star) \leq \frac{2R^2_\mathrm{proj}}{CK}+\frac{B^2 C}{\sqrt{K}}=\mathcal O(K^{-\frac{1}{2}}),
			\end{equation*}
			where $\theta^\star$ denotes the unique minimiser of $G$.
		\end{mythm*}
	\end{frame}

	\begin{frame}[t]{More on the barycentric map}
		\begin{itemize}[<+->]
			\item $\purp{\tilde\rho_K(}\d \purp{y)}\propto \max_i e^{[\hat\theta_K]_i -c_\tau(\rust{\hat x_i^{(n)}},y)}\purp{\gamma(}\d\purp{y)}$ is \textbf{piecewise log-quadratic}
